{{- /* Run the unit cases of assets/cases/*.yaml whose name matches params.test, and whose hugo key, if any, is params.hugo, into results.json, and list the build cases of assets/builds/*.yaml in builds.json. With params.build, run that build case only. */ -}} {{- with site.Params.build -}} {{- $name := . -}} {{- range resources.Match "builds/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}} {{- if eq .name $name -}}{{- partial .partial (partial "test/context.html" .context) -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $filter := site.Params.test -}} {{- $results := slice -}} {{- range resources.Match "cases/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}} {{- if and (or (not $filter) (findRE $filter .name 1)) (or (not .hugo) (eq .hugo site.Params.hugo)) -}} {{- $results = $results | append (partial "test/run.html" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- (resources.FromString "results.json" (jsonify $results)).Publish -}} {{- $builds := slice -}} {{- range resources.Match "builds/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}}{{- $builds = $builds | append . -}}{{- end -}} {{- end -}} {{- (resources.FromString "builds.json" (jsonify $builds)).Publish -}} {{- end -}}
