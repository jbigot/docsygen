{{- /* Run the unit cases of assets/cases/*.yaml whose name matches params.test, into results.json, and list the build cases of assets/builds/*.yaml in builds.json. With params.build, run that build case only. The cases are read as resources: site.Data is deprecated since Hugo 0.156, and hugo.Data is newer than the module's minimum Hugo. */ -}} {{- with site.Params.build -}} {{- $name := . -}} {{- range resources.Match "builds/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}} {{- if eq .name $name -}}{{- partial .partial (partial "test/context.html" .context) -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $filter := site.Params.test -}} {{- $results := slice -}} {{- range resources.Match "cases/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}} {{- if or (not $filter) (findRE $filter .name 1) -}} {{- $results = $results | append (partial "test/run.html" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- (resources.FromString "results.json" (jsonify $results)).Publish -}} {{- $builds := slice -}} {{- range resources.Match "builds/*.yaml" -}}{{- $cases := transform.Unmarshal . -}} {{- range $cases -}}{{- $builds = $builds | append . -}}{{- end -}} {{- end -}} {{- (resources.FromString "builds.json" (jsonify $builds)).Publish -}} {{- end -}}
