{{- /* A text with the words of the refs found in .sources (nodes) linked, as linkifyText does: each ref's text is linked where it first appears after the previous one. The text is escaped. On a class's page (.scope), the class's own name links to its first constructor. */ -}} {{- $text := .text -}} {{- $ctx := .ctx -}} {{- $scope := .scope | default "" -}} {{- $refs := slice -}} {{- range .sources -}} {{- range .children -}}{{- if and (reflect.IsMap .) (eq .tag "ref") -}}{{- $refs = $refs | append . -}}{{- end -}}{{- end -}} {{- end -}} {{- $out := "" -}} {{- range $refs -}} {{- $target := .attributes.refid -}} {{- if and $scope (eq $target $scope) -}} {{- $constructor := partial "docsygen/markup/first-constructor.html" (dict "ctx" $ctx "id" $scope) -}} {{- with $constructor -}}{{- $target = . -}}{{- end -}} {{- end -}} {{- $word := partial "docsygen/text/plain.html" . -}} {{- $url := index $ctx.links $target -}} {{- /* The word where it first appears as a whole word. */ -}} {{- $quoted := replaceRE `([.*+?^${}()|\[\]\\])` `\$1` $word -}} {{- $marked := replaceRE (printf `(^|[^A-Za-z0-9_:])%s([^A-Za-z0-9_:]|$)` $quoted) "${1}\x00${2}" $text 1 -}} {{- if and $url $word (strings.Contains $marked "\x00") -}} {{- $halves := split $marked "\x00" -}} {{- $out = printf "%s%s%s" $out (htmlEscape (index $halves 0)) (htmlEscape $url) (htmlEscape $word) -}} {{- $text = index $halves 1 -}} {{- end -}} {{- end -}} {{- /* .resolve: also link the other words that name a symbol, in the scope or globally. */ -}} {{- if .resolve -}} {{- $m := $ctx.model -}} {{- $prefixes := slice "" -}} {{- with $scope -}} {{- $parts := split (index $m.ids .).name "::" -}} {{- range $i, $_ := $parts -}}{{- $prefixes = slice (printf "%s::" (delimit (first (add $i 1) $parts) "::")) | append $prefixes -}}{{- end -}} {{- end -}} {{- $rest := "" -}} {{- range findRE `[A-Za-z_][A-Za-z0-9_]*(?:::[A-Za-z_][A-Za-z0-9_]*)*|[^A-Za-z_]+` $text -}} {{- $word := . -}} {{- $url := "" -}} {{- if findRE `^[A-Za-z_]` $word 1 -}} {{- range $prefixes -}} {{- if not $url -}}{{- with index $m.names (printf "%s%s" . $word) -}}{{- $url = index $ctx.links (index . 0) -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- $rest = cond (ne $url "") (printf `%s%s` $rest (htmlEscape $url) (htmlEscape $word)) (printf "%s%s" $rest (htmlEscape $word)) -}} {{- end -}} {{- $text = "" -}} {{- $out = printf "%s%s" $out $rest -}} {{- end -}} {{- return (printf "%s%s" $out (htmlEscape $text) | safeHTML) -}}
