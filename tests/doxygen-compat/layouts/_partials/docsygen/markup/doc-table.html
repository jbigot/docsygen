(eq .tag "caption") -}}{{- end -}}{{- end -}} {{- $r := 0 -}} {{- range $n.children -}} {{- if and (reflect.IsMap .) (eq .tag "row") -}} {{- if $markdown -}}{{- else -}}{{- end -}} {{- range .children -}} {{- if and (reflect.IsMap .) (eq .tag "entry") -}} {{- $cell := cond (eq .attributes.thead "yes") "th" "td" -}} {{- $attrs := "" -}} {{- if $markdown -}}{{- $attrs = printf " class=\"markdownTable%s%s\"" (cond (eq $cell "th") "Head" "Body") (index $aligns (.attributes.align | default "") | default "None") -}}{{- end -}} {{- with .attributes.colspan -}}{{- $attrs = printf "%s colspan=\"%s\"" $attrs . -}}{{- end -}} {{- with .attributes.rowspan -}}{{- $attrs = printf "%s rowspan=\"%s\"" $attrs . -}}{{- end -}} {{- printf "<%s%s>" $cell $attrs | safeHTML }}{{ partial "docsygen/markup/doc.html" (merge $sub (dict "node" .)) }}{{ printf "" $cell | safeHTML -}} {{- end -}} {{- end -}} {{- $r = add $r 1 -}} {{- end -}} {{- end -}}
