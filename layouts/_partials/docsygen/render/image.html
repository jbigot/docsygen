{{- /* The source of an image of a description, as {src, ok}: a URL as is; otherwise the file of that name in the XML directory, published at the root of the placement that holds the description's page (whatever baseURL's path), or at the site's root for a part embedded in a page of no version. A missing file warns, naming the image and the page, and gives ok false. Usage: partial "docsygen/render/image.html" (dict "node" $image "ctx" $ctx) */ -}} {{- $name := .node.attributes.name -}} {{- $ctx := .ctx -}} {{- $src := $name -}} {{- $ok := true -}} {{- if not (findRE `^[a-zA-Z][a-zA-Z0-9+.-]*:` $name 1) -}} {{- with resources.Get (path.Join $ctx.model.dir $name) -}} {{- $root := "" -}} {{- with index (partial "docsygen/versions/pages.html" (string $ctx.version)) ($ctx.key | default "") -}} {{- range $p := .page.Ancestors.Reverse | append .page -}}{{- if $p.Params.docsygen.root -}}{{- $root = $p.RelPermalink -}}{{- end -}}{{- end -}} {{- end -}} {{- /* A resource's target path is relative to the site's root, and Hugo puts baseURL's path in front of it: so does RelPermalink. */ -}} {{- $root = strings.TrimPrefix (urls.Parse site.BaseURL).Path $root -}} {{- $src = (resources.Copy (path.Join $root $name) .).RelPermalink -}} {{- else -}} {{- $page := "" -}} {{- with $ctx.page -}}{{- $page = .Path -}}{{- end -}} {{- warnf "docsygen: image %q not found in assets/%s (page %s)" $name $ctx.model.dir $page -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- return dict "src" $src "ok" $ok -}}
