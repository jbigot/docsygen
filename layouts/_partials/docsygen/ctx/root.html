{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* What the renderers need on page .page, for the version of root page .root: {model, paths, topicMembers and variantSeparator (the placement's, docsygen/adapter.html), sourceFormulas (the placement's and the page's, docsygen/ctx/source-formulas.html), anchors: {id: anchor}, links: {id: url} (docsygen/links.html), urls: {page key: url} (docsygen/urls.html), page, version, key}: version is the root's label, key the page's own key when it is a page of that version (docsygen/part.html looks there first), else "". Usage: partial "docsygen/ctx/root.html" (dict "root" $root "page" .Page) */ -}} {{- $root := .root -}} {{- $p := $root.Params.docsygen -}} {{- $model := partial "docsygen/model.html" $p.dir -}} {{- $paths := partial "docsygen/paths.html" (dict "model" $model "pathMap" ($p.pathMap | default slice)) -}} {{- $key := "" -}} {{- $pageFormulas := "" -}} {{- with .page -}}{{- with .Params.docsygen -}}{{- $pageFormulas = .sourceFormulas -}}{{- end -}}{{- end -}} {{- with .page -}}{{- with .Params.docsygen -}}{{- if eq (string .version) (string $p.version) -}}{{- $key = .key | default "" -}}{{- end -}}{{- end -}}{{- end -}} {{- return dict "model" $model "paths" $paths "page" .page "version" $p.version "key" $key "topicMembers" (ne $p.topicMembers false) "variantSeparator" ($p.variantSeparator | default "-") "sourceFormulas" (partial "docsygen/ctx/source-formulas.html" (dict "root" $p.sourceFormulas "page" $pageFormulas)) "anchors" (partialCached "docsygen/ctx/anchors.html" $paths (printf "%s|%s" $p.dir (jsonify ($p.pathMap | default slice)))) "links" (partial "docsygen/links.html" (dict "version" $p.version "paths" $paths)) "urls" (partial "docsygen/urls.html" (string $p.version)) -}}
