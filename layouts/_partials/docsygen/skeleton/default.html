{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The default skeleton of a generated page (docsygen/skeleton.html): each block becomes its shortcode call (docsygen/skeleton/call.html), after a Markdown heading when the block has one ("#" per level; its html as is when the plan gives it, else the text escaped, from the XML node through docsygen/text/markdown.html when the plan gives it; and the anchor and classes as goldmark's attributes, {#… .…}). The title block is left out: Docsy shows the title. A heading block is its heading only. */ -}} {{- $anchors := partialCached "docsygen/ctx/anchors.html" .paths (printf "%s|%s" .dir (jsonify (.pathMap | default slice))) -}} {{- $fixed := index .paths.anchors .entry.key | default dict -}} {{- $out := slice -}} {{- range $b := .blocks -}} {{- if not (in (slice "title") $b.type) -}} {{- with $b.heading -}} {{- $text := "" -}} {{- if .html -}}{{- $text = .html -}} {{- else if ne (.node | default false) false -}}{{- $text = partial "docsygen/text/markdown.html" .node -}} {{- else -}}{{- $text = partial "docsygen/text/md-escape.html" .text -}} {{- end -}} {{- $anchor := index $anchors .anchorKey | default (index $fixed .anchorKey) | default "" -}} {{- if not $anchor -}}{{- warnf "docsygen: no anchor for the heading %q of %s" .text $.entry.key -}}{{- end -}} {{- $attributes := slice -}} {{- with $anchor -}}{{- $attributes = $attributes | append (printf "#%s" .) -}}{{- end -}} {{- range .classes -}}{{- $attributes = $attributes | append (printf ".%s" .) -}}{{- end -}} {{- $out = $out | append (printf "%s %s%s" (strings.Repeat .level "#") $text (cond (gt (len $attributes) 0) (printf " {%s}" (delimit $attributes " ")) "")) -}} {{- end -}} {{- if ne $b.type "heading" -}}{{- $out = $out | append (partial "docsygen/skeleton/call.html" $b) -}}{{- end -}} {{- end -}} {{- end -}} {{- return printf "%s\n" (delimit $out "\n\n") -}}
