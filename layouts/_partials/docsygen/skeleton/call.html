{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The shortcode call that renders a block in a page's Markdown: {{% docsygen/member id=… %}} for a member, {{% docsygen/part id=… part=… name=… %}} for the rest, with the fields of the block's address (docsygen/part/address.html) that are set. Usage, in a skeleton: partial "docsygen/skeleton/call.html" $block */ -}} {{- $a := partial "docsygen/part/address.html" . -}} {{- if eq $a.type "member" -}} {{- return printf "{{%% docsygen/member id=%q %%}}" $a.id -}} {{- end -}} {{- $args := slice -}} {{- range $f := slice "id" "type" "name" -}} {{- with index $a $f -}} {{- $args = $args | append (printf "%s=%q" (index (dict "id" "id" "type" "part" "name" "name") $f) .) -}} {{- end -}} {{- end -}} {{- return printf "{{%% docsygen/part %s %%}}" (delimit $args " ") -}}
