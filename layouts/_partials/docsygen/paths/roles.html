{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The content units of a model, independent of any URL scheme, in this order: {role: root} {role: compound, id} every compound that has a page in Doxygen's HTML, followed by {role: all-members, id} its list of all members (class-likes with listofallmembers) {role: source, id} and its source listing (files Doxygen writes one for) {role: tree, name} annotated, hierarchy, namespaces, concepts, files, pages, topics, examples Only what Doxygen 1.18 generates is listed (src/index.cpp, classdef.cpp, filedef.cpp, namespacedef.cpp), honouring the Doxyfile options that decide it: those of docsygen/paths/has-page.html for the compounds, OPTIMIZE_OUTPUT_FOR_C, SOURCE_BROWSER, VERBATIM_HEADERS. Usage: partial "docsygen/paths/roles.html" $model */ -}} {{- $m := . -}} {{- $yes := newScratch -}} {{- range $k, $v := dict "OPTIMIZE_OUTPUT_FOR_C" "NO" "SOURCE_BROWSER" "NO" "VERBATIM_HEADERS" "YES" -}} {{- $yes.Set $k (eq (index ((index $m.options $k) | default (slice $v)) 0) "YES") -}} {{- end -}} {{- $classLike := slice "class" "struct" "union" "interface" "protocol" "category" "exception" -}} {{- $page := partial "docsygen/paths/has-page.html" $m -}} {{- $roles := slice (dict "role" "root") -}} {{- $count := newScratch -}} {{- range $m.order -}} {{- $id := . -}} {{- $e := index $m.ids . -}} {{- if index $page $id -}} {{- $roles = $roles | append (dict "role" "compound" "id" $id) -}} {{- $count.Add $e.kind 1 -}} {{- end -}} {{- if and (in $classLike $e.kind) (index $page $id) (not ($yes.Get "OPTIMIZE_OUTPUT_FOR_C")) -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "listofallmembers") .children -}} {{- $roles = $roles | append (dict "role" "all-members" "id" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $e.kind "file" -}} {{- /* filedef.cpp, generateSourceFile: not a documentation file, and SOURCE_BROWSER, or VERBATIM_HEADERS for a header or a file that others include. */ -}} {{- $ext := lower (path.Ext $e.name) -}} {{- $header := in (slice ".h" ".hh" ".hxx" ".hpp" ".h++" ".idl" ".ddl" ".pidl" ".ice") $ext -}} {{- $doc := in (slice ".doc" ".txt" ".dox" ".md" ".markdown") $ext -}} {{- $listing := false -}} {{- $included := false -}} {{- range $e.node.children -}} {{- if not (reflect.IsMap .) -}} {{- else if eq .tag "programlisting" -}}{{- $listing = true -}} {{- else if eq .tag "includedby" -}}{{- $included = true -}} {{- end -}} {{- end -}} {{- if and $listing (not $doc) (or ($yes.Get "SOURCE_BROWSER") (and ($yes.Get "VERBATIM_HEADERS") (or $header $included))) -}} {{- $roles = $roles | append (dict "role" "source" "id" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* The lists. */ -}} {{- $hasClasses := false -}} {{- range $classLike -}}{{- if $count.Get . -}}{{- $hasClasses = true -}}{{- end -}}{{- end -}} {{- $hierarchy := false -}} {{- range $m.order -}} {{- $e := index $m.ids . -}} {{- if and (in $classLike $e.kind) (index $page .) -}} {{- $derived := false -}} {{- $root := true -}} {{- range $e.node.children -}} {{- if not (reflect.IsMap .) -}} {{- else if eq .tag "derivedcompoundref" -}}{{- $derived = true -}} {{- else if and (eq .tag "basecompoundref") .attributes.refid (index $page .attributes.refid) -}}{{- $root = false -}} {{- end -}} {{- end -}} {{- if and $derived $root -}}{{- $hierarchy = true -}}{{- end -}} {{- end -}} {{- end -}} {{- $pages := 0 -}} {{- range $m.order -}}{{- if and (eq (index $m.ids .).kind "page") (ne . "indexpage") -}}{{- $pages = add $pages 1 -}}{{- end -}}{{- end -}} {{- range $name, $present := dict "annotated" $hasClasses "hierarchy" $hierarchy "namespaces" ($count.Get "namespace") "concepts" ($count.Get "concept") "files" ($count.Get "file") "pages" $pages "topics" ($count.Get "group") "examples" ($count.Get "example") -}} {{- if $present -}}{{- $roles = $roles | append (dict "role" "tree" "name" $name) -}}{{- end -}} {{- end -}} {{- return $roles -}}
