{{- /* The content units of a model, independent of any URL scheme, in this order: {role: root} {role: compound, id} every compound that has a page in Doxygen's HTML, followed by {role: all-members, id} its list of all members (class-likes with listofallmembers) {role: source, id} and its source listing (files Doxygen writes one for) {role: tree, name} annotated, hierarchy, namespaces, concepts, files, pages, topics, examples {role: index, name, filter, letter} classes, class-members, namespace-members and file-members, by filter (all, functions, variables, typedefs, enums, enumvalues, properties, events, related, macros), and by first letter when the filter has more than 200 members (letter is "" otherwise) Only what Doxygen 1.15 generates is listed (src/index.cpp, classdef.cpp, filedef.cpp, namespacedef.cpp), honouring the Doxyfile options that decide it: those of docsygen/paths/has-page.html for the compounds, OPTIMIZE_OUTPUT_FOR_C, SOURCE_BROWSER, VERBATIM_HEADERS, and those of docsygen/paths/index-entries.html for the indexes. Usage: partial "docsygen/paths/roles.html" $model */ -}} {{- $m := . -}} {{- $yes := newScratch -}} {{- range $k, $v := dict "OPTIMIZE_OUTPUT_FOR_C" "NO" "SOURCE_BROWSER" "NO" "VERBATIM_HEADERS" "YES" -}} {{- $yes.Set $k (eq (index ((index $m.options $k) | default (slice $v)) 0) "YES") -}} {{- end -}} {{- $classLike := slice "class" "struct" "union" "interface" "protocol" "category" "exception" -}} {{- $page := partial "docsygen/paths/has-page.html" $m -}} {{- $roles := slice (dict "role" "root") -}} {{- $count := newScratch -}} {{- range $m.order -}} {{- $id := . -}} {{- $e := index $m.ids . -}} {{- if index $page $id -}} {{- $roles = $roles | append (dict "role" "compound" "id" $id) -}} {{- $count.Add $e.kind 1 -}} {{- end -}} {{- if and (in $classLike $e.kind) (index $page $id) (not ($yes.Get "OPTIMIZE_OUTPUT_FOR_C")) -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "listofallmembers") .children -}} {{- $roles = $roles | append (dict "role" "all-members" "id" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $e.kind "file" -}} {{- /* filedef.cpp, generateSourceFile: not a documentation file, and SOURCE_BROWSER, or VERBATIM_HEADERS for a header or a file that others include. */ -}} {{- $ext := lower (path.Ext $e.name) -}} {{- $header := in (slice ".h" ".hh" ".hxx" ".hpp" ".h++" ".idl" ".ddl" ".pidl" ".ice") $ext -}} {{- $doc := in (slice ".doc" ".txt" ".dox" ".md" ".markdown") $ext -}} {{- $listing := false -}} {{- $included := false -}} {{- range $e.node.children -}} {{- if not (reflect.IsMap .) -}} {{- else if eq .tag "programlisting" -}}{{- $listing = true -}} {{- else if eq .tag "includedby" -}}{{- $included = true -}} {{- end -}} {{- end -}} {{- if and $listing (not $doc) (or ($yes.Get "SOURCE_BROWSER") (and ($yes.Get "VERBATIM_HEADERS") (or $header $included))) -}} {{- $roles = $roles | append (dict "role" "source" "id" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* The lists. */ -}} {{- $hasClasses := false -}} {{- range $classLike -}}{{- if $count.Get . -}}{{- $hasClasses = true -}}{{- end -}}{{- end -}} {{- $hierarchy := false -}} {{- range $m.order -}} {{- $e := index $m.ids . -}} {{- if and (in $classLike $e.kind) (index $page .) -}} {{- $derived := false -}} {{- $root := true -}} {{- range $e.node.children -}} {{- if not (reflect.IsMap .) -}} {{- else if eq .tag "derivedcompoundref" -}}{{- $derived = true -}} {{- else if and (eq .tag "basecompoundref") .attributes.refid (index $page .attributes.refid) -}}{{- $root = false -}} {{- end -}} {{- end -}} {{- if and $derived $root -}}{{- $hierarchy = true -}}{{- end -}} {{- end -}} {{- end -}} {{- $pages := 0 -}} {{- range $m.order -}}{{- if and (eq (index $m.ids .).kind "page") (ne . "indexpage") -}}{{- $pages = add $pages 1 -}}{{- end -}}{{- end -}} {{- range $name, $present := dict "annotated" $hasClasses "hierarchy" $hierarchy "namespaces" ($count.Get "namespace") "concepts" ($count.Get "concept") "files" ($count.Get "file") "pages" $pages "topics" ($count.Get "group") "examples" ($count.Get "example") -}} {{- if $present -}}{{- $roles = $roles | append (dict "role" "tree" "name" $name) -}}{{- end -}} {{- end -}} {{- /* The indexes, from the member lists of index.xml: each class, namespace or file lists its own members (index.cpp, add*MemberNameToIndex). */ -}} {{- if $hasClasses -}}{{- $roles = $roles | append (dict "role" "index" "name" "classes" "filter" "all" "letter" "") -}}{{- end -}} {{- $entries := partial "docsygen/paths/index-entries.html" $m -}} {{- range slice "class-members" "namespace-members" "file-members" -}} {{- $index := . -}} {{- $list := index $entries $index -}} {{- range slice "all" "functions" "variables" "typedefs" "enums" "enumvalues" "properties" "events" "related" "macros" -}} {{- $filter := . -}} {{- $letters := newScratch -}} {{- $n := 0 -}} {{- range where $list "filters" "intersect" (slice $filter) -}}{{- $letters.Set .letter true -}}{{- $n = add $n 1 -}}{{- end -}} {{- if gt $n 200 -}} {{- /* index.cpp sorts the letters as bytes, as the keys of this map are sorted. */ -}} {{- range $letter, $_ := $letters.Values -}} {{- $roles = $roles | append (dict "role" "index" "name" $index "filter" $filter "letter" $letter) -}} {{- end -}} {{- else if $n -}} {{- $roles = $roles | append (dict "role" "index" "name" $index "filter" $filter "letter" "") -}} {{- end -}} {{- end -}} {{- end -}} {{- return $roles -}}
