{{- /* The content units of a model, independent of any URL scheme, in this order: {role: root} {role: compound, id} every compound that has a page in Doxygen's HTML, followed by {role: all-members, id} its list of all members (class-likes with listofallmembers) {role: source, id} and its source listing (files Doxygen writes one for) {role: tree, name} annotated, hierarchy, namespaces, concepts, files, pages, topics, examples {role: index, name, filter, letter} classes, class-members, namespace-members and file-members, by filter (all, functions, variables, typedefs, enums, enumvalues, properties, events, related, macros), and by first letter when the filter has more than 200 members (letter is "" otherwise) Only what Doxygen 1.15 generates is listed (src/index.cpp, classdef.cpp, filedef.cpp, namespacedef.cpp), honouring the Doxyfile options that decide it: SHOW_FILES, INLINE_SIMPLE_STRUCTS, EXTRACT_ANON_NSPACES, OPTIMIZE_OUTPUT_FOR_C, SOURCE_BROWSER, VERBATIM_HEADERS, IGNORE_PREFIX, and those of docsygen/model/linkable.html for the members of the indexes. (in (slice "innerclass" "innernamespace") .tag) -}} {{- if eq $e.kind "group" -}} {{- $grouped.Set .attributes.refid true -}} {{- else if ne $e.kind "file" -}} {{- $parent.Set .attributes.refid $e.id -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Which compounds have a page. (eq .tag "memberdef") -}} {{- $type := "" -}} {{- range .children -}}{{- if and (reflect.IsMap .) (eq .tag "type") -}}{{- $type = partial "docsygen/text/plain.html" . -}}{{- end -}}{{- end -}} {{- $field := and (eq .attributes.kind "variable") (ne .attributes.static "yes") -}} {{- $alias := and (eq .attributes.kind "typedef") (not (strings.Contains $type ")(")) -}} {{- if not (and (eq .attributes.prot "public") (or $field $alias)) -}}{{- $simple = false -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $simple -}} {{- $linkable := false -}} {{- with $parent.Get .id -}} {{- $linkable = $page.Get . -}} {{- else -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) "indexpage") -}}{{- $pages = add $pages 1 -}}{{- end -}}{{- end -}} {{- range $name, $present := dict "annotated" $hasClasses "hierarchy" $hierarchy "namespaces" ($count.Get "namespace") "concepts" ($count.Get "concept") "files" ($count.Get "file") "pages" $pages "topics" ($count.Get "group") "examples" ($count.Get "example") -}} {{- if $present -}}{{- $roles = $roles | append (dict "role" "tree" "name" $name) -}}{{- end -}} {{- end -}} {{- /* The indexes, from the member lists of index.xml: each class, namespace or file lists its own members (index.cpp, add*MemberNameToIndex). */ -}} {{- if $hasClasses -}}{{- $roles = $roles | append (dict "role" "index" "name" "classes" "filter" "all" "letter" "") -}}{{- end -}} {{- $filters := dict "function" "functions" "slot" "functions" "signal" "functions" "variable" "variables" "typedef" "typedefs" "enum" "enums" "enumvalue" "enumvalues" "property" "properties" "event" "events" "friend" "related" "define" "macros" -}} {{- $indexes := newScratch -}} {{- with resources.Get (printf "%s/index.xml" $m.dir) -}} {{- range (partial "docsygen/xml/unmarshal.html" .).children -}} {{- if and (reflect.IsMap .) (eq .tag "compound") -}} {{- $kind := .attributes.kind -}} {{- $cid := .attributes.refid -}} {{- $index := "" -}} {{- if and (in $classLike $kind) (not (strings.Contains (index $m.ids $cid).name "@")) -}} {{- $index = "class-members" -}} {{- else if and (eq $kind "namespace") ($page.Get $cid) -}} {{- $index = "namespace-members" -}} {{- else if and (eq $kind "file") ($page.Get $cid) -}} {{- $index = "file-members" -}} {{- end -}} {{- if $index -}} {{- range .children -}} {{- if and (reflect.IsMap .) (eq .tag "member") -}} {{- $mid := .attributes.refid -}} {{- $mkind := .attributes.kind -}} {{- $name := "" -}} {{- range .children -}}{{- if and (reflect.IsMap .) (eq .tag "name") -}}{{- $name = delimit .children "" -}}{{- end -}}{{- end -}} {{- range $prefixes -}}{{- if and . (hasPrefix $name .) (gt (len $name) (len .)) -}}{{- $name = strings.TrimPrefix . $name -}}{{- end -}}{{- end -}} {{- $letter := lower (substr $name 0 1) -}} {{- $strong := false -}} {{- if eq $mkind "enumvalue" -}} {{- with index $m.ids $mid -}}{{- $strong = eq (index $m.ids .enum).node.attributes.strong "yes" -}}{{- end -}} {{- end -}} {{- $filter := index $filters $mkind -}} {{- if and (eq $filter "macros") (ne $index "file-members") -}}{{- $filter = "" -}}{{- end -}} {{- $linkable := partial "docsygen/model/linkable.html" (dict "model" $m "id" $mid) -}} {{- if and $letter (not $strong) $linkable -}} {{- $indexes.SetInMap $index (printf "all|%s|%s" $letter $mid) true -}} {{- if $filter -}}{{- $indexes.SetInMap $index (printf "%s|%s|%s" $filter $letter $mid) true -}}{{- end -}} {{- end -}} {{- /* index.xml lists enum values under namespaces, not under files: take them from the enum. */ -}} {{- if and (eq $mkind "enum") $linkable (ne (index $m.ids $mid).node.attributes.strong "yes") -}} {{- range (index $m.ids $mid).node.children -}} {{- if and (reflect.IsMap .) (eq .tag "enumvalue") -}} {{- $vid := .attributes.id -}} {{- $vname := "" -}} {{- range .children -}}{{- if and (reflect.IsMap .) (eq .tag "name") -}}{{- $vname = delimit .children "" -}}{{- end -}}{{- end -}} {{- range $prefixes -}}{{- if and . (hasPrefix $vname .) (gt (len $vname) (len .)) -}}{{- $vname = strings.TrimPrefix . $vname -}}{{- end -}}{{- end -}} {{- with lower (substr $vname 0 1) -}} {{- $indexes.SetInMap $index (printf "all|%s|%s" . $vid) true -}} {{- $indexes.SetInMap $index (printf "enumvalues|%s|%s" . $vid) true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range slice "class-members" "namespace-members" "file-members" -}} {{- $index := . -}} {{- $entries := $indexes.Get $index | default dict -}} {{- range slice "all" "functions" "variables" "typedefs" "enums" "enumvalues" "properties" "events" "related" "macros" -}} {{- $filter := . -}} {{- $letters := newScratch -}} {{- $n := 0 -}} {{- range $key, $_ := $entries -}} {{- $parts := split $key "|" -}} {{- if eq (index $parts 0) $filter -}}{{- $letters.Set (index $parts 1) true -}}{{- $n = add $n 1 -}}{{- end -}} {{- end -}} {{- if gt $n 200 -}} {{- /* index.cpp sorts the letters as bytes, as the keys of this map are sorted. */ -}} {{- range $letter, $_ := $letters.Values -}} {{- $roles = $roles | append (dict "role" "index" "name" $index "filter" $filter "letter" $letter) -}} {{- end -}} {{- else if $n -}} {{- $roles = $roles | append (dict "role" "index" "name" $index "filter" $filter "letter" "") -}} {{- end -}} {{- end -}} {{- end -}} {{- return $roles -}}
