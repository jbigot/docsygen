{{- /* The Markdown content of a generated page, from its plan: each block becomes a {{< docsygen/block n=K >}} call, after a Markdown heading when the block has one ("#" per level, the text escaped, from the XML node through docsygen/text/markdown.html when the plan gives it, and the anchor as {#…}). The title and summary-links blocks are left out: Docsy shows the title and the page's table of contents. Usage: partial "docsygen/skeleton.html" (dict "entry" $page "paths" $paths "dir" $dir "pathMap" $pathMap) */ -}} {{- $m := partial "docsygen/model.html" .dir -}} {{- $anchors := partialCached "docsygen/ctx/anchors.html" .paths (printf "%s|%s" .dir (jsonify (.pathMap | default slice))) -}} {{- $fixed := index .paths.anchors .entry.key | default dict -}} {{- $ctx := dict "model" $m "paths" .paths -}} {{- $out := slice -}} {{- range $k, $b := partial "docsygen/plan.html" (dict "entry" .entry "ctx" $ctx) -}} {{- if not (in (slice "title" "summary-links") $b.type) -}} {{- with $b.heading -}} {{- $text := cond (ne (.node | default false) false) (partial "docsygen/text/markdown.html" (.node | default dict)) (partial "docsygen/text/md-escape.html" .text) -}} {{- $anchor := index $anchors .anchorKey | default (index $fixed .anchorKey) | default "" -}} {{- if not $anchor -}}{{- warnf "docsygen: no anchor for the heading %q of %s" .text $.entry.key -}}{{- end -}} {{- $out = $out | append (printf "%s %s%s" (strings.Repeat .level "#") $text (cond (ne $anchor "") (printf " {#%s}" $anchor) "")) -}} {{- end -}} {{- $out = $out | append (printf "{{< docsygen/block n=%d >}}" $k) -}} {{- end -}} {{- end -}} {{- return printf "%s\n" (delimit $out "\n\n") -}}
