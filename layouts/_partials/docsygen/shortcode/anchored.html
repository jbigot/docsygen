"ctx" $ctx "id" $id) */ -}} {{- $page := .shortcode.Page -}} {{- $anchor := index .ctx.anchors .id -}} {{- $taken := printf "docsygen-anchor|%s|%s" .ctx.version .id -}} {{- $raw := $page.RawContent -}} {{- $headed := or (strings.Contains $raw (printf "{#%s}" $anchor)) (strings.Contains $raw (printf "{#%s " $anchor)) -}} {{- /* The taker's ordinal plus one: the first shortcode of a page is 0, which reads as none. */ -}} {{- $mine := add .shortcode.Ordinal 1 -}} {{- if not (or $headed ($page.Store.Get $taken)) -}} {{- $page.Store.Set $taken $mine -}} {{- end -}} {{- return eq ($page.Store.Get $taken) $mine -}}
