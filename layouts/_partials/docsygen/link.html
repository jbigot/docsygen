{{- /* A link of the link render hook whose destination is doxygen:…, resolved by docsygen/resolve.html: an empty text becomes the symbol's name as code, or the page's or group's title. A destination that resolves to nothing stops the build. A link to a version that development mode skips renders its text, or the name as code, without a link, and warns. Usage: partial "docsygen/link.html" . (the render hook's context) */ -}} {{- $r := partial "docsygen/resolve.html" (dict "target" .Destination "page" .Page) -}} {{- if $r.skipped -}} {{- warnidf "docsygen-version-skipped" "docsygen: version %q is not built here: %s links to it as text" $r.skipped .Page.Path -}} {{- .Text | default (printf "%s" (htmlEscape $r.name) | safeHTML) -}} {{- else if not $r.ok -}} {{- errorf "docsygen: %s: cannot resolve %q" .Page.Path .Destination -}} {{- else -}} {{- $text := .Text -}} {{- if not $text -}} {{- $text = cond (ne $r.title "") (htmlEscape $r.title) (printf "%s" (htmlEscape $r.name)) | safeHTML -}} {{- end -}} {{ $text }} {{- end -}}
