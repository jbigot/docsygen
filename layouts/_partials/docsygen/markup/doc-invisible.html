{{- /* Whether a description node shows nothing in HTML (htmldocvisitor.cpp, isInvisibleNode): whitespace, images for other outputs, and the other outputs' verbatims. */ -}} {{- $n := . -}} {{- $invisible := false -}} {{- if reflect.IsMap $n -}} {{- $invisible = in (slice "latexonly" "manonly" "rtfonly" "xmlonly" "docbookonly") $n.tag -}} {{- if eq $n.tag "image" -}}{{- $invisible = ne $n.attributes.type "html" -}}{{- end -}} {{- else -}} {{- $invisible = not (trim $n " \t\n\r") -}} {{- end -}} {{- return $invisible -}}
