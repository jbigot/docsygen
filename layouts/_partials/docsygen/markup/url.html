{{- /* The URL of target .id in the version of .ctx (its links), or "". A target that no page of the version holds warns, naming the page and the target: the caller writes its text alone. Usage: partial "docsygen/markup/url.html" (dict "ctx" $ctx "id" $id) */ -}} {{- $ctx := .ctx -}} {{- $url := "" -}} {{- if and .id (isset $ctx.links .id) -}} {{- $url = index $ctx.links .id -}} {{- if eq $url "" -}} {{- $name := .id -}} {{- with index $ctx.model.ids .id -}}{{- $name = .name | default $name -}}{{- end -}} {{- $page := "" -}} {{- with $ctx.page -}}{{- $page = .Path -}}{{- end -}} {{- warnidf "docsygen-unplaced" "docsygen: %s links to %s, which no page of version %s holds: rendered as text" $page $name $ctx.version -}} {{- end -}} {{- end -}} {{- return $url -}}
