{{- /* A text with the words of the refs found in .sources (nodes) linked, as linkifyText does: each ref's text is linked where it first appears after the previous one. The text is escaped. On a class's page (.scope), the class's own name links to its first constructor. Doxygen also links other words that name a symbol, which the XML doesn't link (an XML gap). */ -}} {{- $text := .text -}} {{- $ctx := .ctx -}} {{- $scope := .scope | default "" -}} {{- $refs := slice -}} {{- range .sources -}} {{- range .children -}}{{- if and (reflect.IsMap .) (eq .tag "ref") -}}{{- $refs = $refs | append . -}}{{- end -}}{{- end -}} {{- end -}} {{- $out := "" -}} {{- range $refs -}} {{- $target := .attributes.refid -}} {{- if and $scope (eq $target $scope) -}} {{- $constructor := partial "docsygen/markup/first-constructor.html" (dict "ctx" $ctx "id" $scope) -}} {{- with $constructor -}}{{- $target = . -}}{{- end -}} {{- end -}} {{- $word := partial "docsygen/text/plain.html" . -}} {{- $url := index $ctx.links $target -}} {{- /* The word where it first appears as a whole word. */ -}} {{- $quoted := replaceRE `([.*+?^${}()|\[\]\\])` `\$1` $word -}} {{- $marked := replaceRE (printf `(^|[^A-Za-z0-9_:])%s([^A-Za-z0-9_:]|$)` $quoted) "${1}\x00${2}" $text 1 -}} {{- if and $url $word (strings.Contains $marked "\x00") -}} {{- $halves := split $marked "\x00" -}} {{- $out = printf "%s%s%s" $out (htmlEscape (index $halves 0)) (htmlEscape $url) (htmlEscape $word) -}} {{- $text = index $halves 1 -}} {{- end -}} {{- end -}} {{- return (printf "%s%s" $out (htmlEscape $text) | safeHTML) -}}
