{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* What a namespace, topic, file, directory or class page lists of its declarations (docsygen/plan/list.html, which docsygen/plan/compound.html puts in the block as list): the direct sub-namespaces or sub-topics first, each on a line of its own, then a line naming the namespace or topic, then the page's buckets (docsygen/markup/list-buckets.html), sub-namespaces and sub-topics last with their content nested. On a namespace's list, a sub-namespace's line reads "namespace ::", the line naming the namespace "namespace ::::" (the root's "namespace ::"). On a topic's list, a sub-topic's line reads "topic › <sub-topic's title>", the line naming the topic "topic <title>". The line naming it comes when it has entries of its own, and on the root, when it lists namespaces too. On its own page, a namespace's or topic's list gives each nested namespace's or sub-topic's line its anchor (namespace:<id> or topic:<id>, docsygen/paths/scheme.html), and a line at the top links there. Away from its page (docsygen/part.html), where the scheme can't keep those anchors unique, the nested lines have none and a line at the top links to the namespace's or sub-topic's own page. */ -}} {{- $ctx := .ctx -}} {{- $b := .block -}} {{- $list := $b.list -}} {{- $anchors := dict -}} {{- if and (not $b.away) $ctx.paths $ctx.key -}} {{- $anchors = index $ctx.paths.anchors $ctx.key | default dict -}} {{- end -}} {{- /* A line at the top: the word in its kind's colour, then the name, linked when .url is not "". Codes in a span inside a div: Docsy gives the inline-code background to code directly in a paragraph or a list item. */ -}} {{- $line := `<div class="docsygen-list-scope"><span class="docsygen-entity"><code class="docsygen-kind docsygen-kind-%s fw-bold">%s</code> %s</span></div>` -}} {{- $lines := "" -}} {{- range $list.subs -}} {{- $sub := . -}} {{- range $list.buckets -}} {{- range .entries -}} {{- if eq .id $sub -}} {{- $name := printf "%s › %s" $list.topic .name -}} {{- if eq .word "namespace" -}}{{- $name = printf "::%s" (index $ctx.model.ids .id).name -}}{{- end -}} {{- $url := "" -}} {{- with index $anchors (printf "%s:%s" .word .id) -}}{{- $url = printf "%s#%s" (index ($ctx.urls | default dict) $ctx.key | default "") . -}}{{- end -}} {{- $url = $url | default (partial "docsygen/markup/url.html" (dict "ctx" $ctx "id" .id)) -}} {{- $code := cond (ne $url "") (printf `<a href="%s"><code>%s</code></a>` (htmlEscape $url) (htmlEscape $name)) (printf `<code class="text-reset">%s</code>` (htmlEscape $name)) -}} {{- $lines = printf "%s%s" $lines (printf $line .word .word $code) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt ($list.own | default 0) 0 -}} {{- with $list.namespace -}} {{- $name := replace .
