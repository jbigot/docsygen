The fence is longer than any run of backticks in the source. Usage: partial "docsygen/markup/mermaid-block.html" (dict "source" $source "ctx" $ctx) */ -}} {{- $source := .source -}} {{- $ticks := 3 -}} {{- range findRE "`+" $source -}}{{- $ticks = math.Max $ticks (add (len .) 1) -}}{{- end -}} {{- $fence := strings.Repeat (int $ticks) "`" -}} {{- return .ctx.page.RenderString (dict "display" "block") (printf "%smermaid\n%s\n%s" $fence $source $fence) -}}
