{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The buckets of a list (docsygen/plan/list.html), one
    each, of lines (docsygen/markup/list-line.html), with a gap between two; a bucket that starts a label follows that label, as a small paragraph: static, protected, private, package, related, a member group's heading, then its description, or "inherited from" its base, linked, after the label of the inherited members when they are not public. .anchors, {namespace: or topic:: anchor} (default none), gives a nested namespace's or sub-topic's line its anchor. Usage: partial "docsygen/markup/list-buckets.html" (dict "ctx" $ctx "buckets" $buckets "page" $pageId "anchors" $anchors) */ -}} {{- $ctx := .ctx -}} {{- $m := $ctx.model -}} {{- $page := .page -}} {{- $anchors := .anchors | default dict -}} {{- $out := "" -}} {{- $prev := "" -}} {{- range .buckets -}} {{- if and .label (ne .label $prev) -}} {{- $label := "" -}} {{- if in (slice "static" "protected" "private" "package" "related") .label -}} {{- $label = T (printf "docsygen-list-%s" .label) -}} {{- else if findRE `^inherited(-[a-z]+)?:` .label -}} {{- $parts := split .label ":" -}} {{- $base := index $parts 1 -}} {{- $baseName := htmlEscape (partial "docsygen/plan/display-name.html" (dict "model" $m "id" $base)) -}} {{- $url := partial "docsygen/markup/url.html" (dict "ctx" $ctx "id" $base) -}} {{- with $url -}}{{- $baseName = printf `%s` (htmlEscape .) $baseName -}}{{- end -}} {{- $label = T "docsygen-list-inherited-from" (dict "base" $baseName) -}} {{- with strings.TrimPrefix "inherited-" (index $parts 0) | strings.TrimPrefix "inherited" -}} {{- $label = T "docsygen-list-inherited-labelled" (dict "label" (T (printf "docsygen-list-%s" .)) "base" $baseName) -}} {{- end -}} {{- else -}} {{- $label = htmlEscape .label -}} {{- end -}} {{- $out = printf `%s

    %s

    ` $out $label -}} {{- with .description -}} {{- $out = printf `%s
    %s
    ` $out (strings.TrimSpace (string (partial "docsygen/markup/doc.html" (dict "node" . "ctx" $ctx "single" true)))) -}} {{- end -}} {{- end -}} {{- $prev = .label -}} {{- $lines := "" -}} {{- range .entries -}}{{- $lines = printf "%s%s" $lines (partial "docsygen/markup/list-line.html" (dict "ctx" $ctx "entry" . "page" $page "anchors" $anchors)) -}}{{- end -}} {{- $out = printf `%s
      %s
    ` $out $lines -}} {{- end -}} {{- return safeHTML $out -}}
