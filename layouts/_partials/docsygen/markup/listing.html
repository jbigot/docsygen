{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* A program listing as a fenced code block, rendered by .ctx.page.RenderString: Hugo's highlighter, the site's markup.highlight settings and any render-codeblock hook apply. The language is .lang, or the extension of the listing's filename, else text. A .numbered listing, and an example's own listing, has line numbers from its first codeline's lineno (1 without), each line the anchor l- (Chroma's lineanchors), which links to a line target (docsygen/plan/examples.html, docsygen/markup/definition.html); a site whose settings or hook drop them gets a warning. Code links are not kept. Usage: partial "docsygen/markup/listing.html" (dict "node" $programlisting "ctx" $ctx "numbered" false "lang" "") */ -}} {{- $ctx := .ctx -}} {{- $lang := .lang | default (strings.TrimPrefix "." (path.Ext (.node.attributes.filename | default ""))) | default "text" -}} {{- $numbered := .numbered | default false -}} {{- with $ctx.key -}} {{- range where $ctx.paths.pages "key" . -}} {{- range .roles -}} {{- $page := and (eq .role "compound") (index $ctx.model.ids .id) -}} {{- if and $page (eq $page.kind "example") (eq $.node.attributes.filename $page.name) -}}{{- $numbered = true -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* No line anchors in a copy that mustn't repeat the page's ids (.ctx.noids). */ -}} {{- if $ctx.noids -}}{{- $numbered = false -}}{{- end -}} {{- /* The code, a line per codeline; a numbered listing is padded with empty lines where the XML omits some (the lines of documentation comments), so that every line has its own number. */ -}} {{- $first := 0 -}} {{- $lines := slice -}} {{- range .node.children -}} {{- if and (reflect.IsMap .) (eq .tag "codeline") -}} {{- $n := int (.attributes.lineno | default 0) -}} {{- if not $first -}}{{- $first = $n | default 1 -}}{{- end -}} {{- if and $numbered $n (gt (sub $n $first (len $lines)) 0) -}} {{- range seq (sub $n $first (len $lines)) -}}{{- $lines = $lines | append "" -}}{{- end -}} {{- end -}} {{- $lines = $lines | append (partial "docsygen/render/listing-text.html" .) -}} {{- end -}} {{- end -}} {{- $first = $first | default 1 -}} {{- $code := delimit $lines "\n" -}} {{- $info := $lang -}} {{- if $numbered -}}{{- $info = printf `%s {linenos=inline, anchorlinenos=true, lineanchors="l", linenostart=%d}` $lang $first -}}{{- end -}} {{- $html := partial "docsygen/markup/fence.html" (dict "info" $info "code" $code "ctx" $ctx) -}} {{- if and $numbered (not (strings.Contains (string $html) (printf `id="l-%d"` $first))) -}} {{- warnidf "docsygen-listing-anchors" "docsygen: %s: a listing has no line anchors (id=\"l-%d\"), which links to its lines target: check markup.highlight (lineNos, anchorLineNos, lineAnchors, codeFences) and the site's render-codeblock hook" $ctx.page.Path $first -}} {{- end -}} {{- return $html -}}
