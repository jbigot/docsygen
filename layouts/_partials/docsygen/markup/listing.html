{{- /* A program listing as a highlighted pre, a span.docsygen-line per codeline, each highlighted on its own (transform.Highlight, inline, with the site's highlighting settings): a token that spans lines, such as a block comment, is highlighted line by line. The language is .lang, or the extension of the listing's filename, else text. A .numbered line has the anchor l (lNNNNN) and its number in data-line, which the stylesheet shows; an example's own listing starts the line of each of its anchors (docsygen/plan/examples.html) with it. Code links are not kept. Usage: partial "docsygen/markup/listing.html" (dict "node" $programlisting "ctx" $ctx "numbered" false "lang" "") */ -}} {{- $ctx := .ctx -}} {{- $lang := .lang | default (strings.TrimPrefix "." (path.Ext (.node.attributes.filename | default ""))) | default "text" -}} {{- $anchors := slice -}} {{- with $ctx.page -}} {{- range where $ctx.paths.pages "key" (.Params.docsygen.key | default "") -}} {{- range .roles -}} {{- $page := and (eq .role "compound") (index $ctx.model.ids .id) -}} {{- if and $page (eq $page.kind "example") (eq $.node.attributes.filename $page.name) -}} {{- $anchors = index (partial "docsygen/plan/examples.html" (dict "model" $ctx.model)).anchors $page.id | default slice -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- define "_partials/docsygen/markup/inline/listing-refs.html" -}} {{- /* The number of refs under . */ -}} {{- $n := 0 -}} {{- range .children -}} {{- if reflect.IsMap . -}}{{- $n = add $n (cond (eq .tag "ref") 1 (partial "docsygen/markup/inline/listing-refs.html" .)) -}}{{- end -}} {{- end -}} {{- return $n -}} {{- end -}} {{- $r := 0 -}} {{- $lines := slice -}} {{- range .node.children -}} {{- if and (reflect.IsMap .) (eq .tag "codeline") -}} {{- $text := partial "docsygen/render/listing-text.html" . -}} {{- $code := "" -}} {{- with $text -}} {{- $code = string (transform.Highlight . $lang (dict "hl_inline" true)) -}} {{- /* Chroma ends a line comment's token with a newline, which one codeline can't hold. */ -}} {{- $code = replace (replaceRE `^]*>` "" (strings.TrimSuffix "" $code)) "\n" "" -}} {{- end -}} {{- $marks := "" -}} {{- $refs := partial "docsygen/markup/inline/listing-refs.html" . -}} {{- range seq $refs -}} {{- with and (lt $r (len $anchors)) (index $anchors $r) -}}{{- $marks = printf `%s` $marks . -}}{{- end -}} {{- $r = add $r 1 -}} {{- end -}} {{- $attrs := "" -}} {{- if $.numbered -}}{{- $attrs = printf ` id="l%05d" data-line="%d"` (int .attributes.lineno) (int .attributes.lineno) -}}{{- end -}} {{- $lines = $lines | append (printf `%s%s` $attrs $marks $code) -}} {{- end -}} {{- end -}} 
{{ delimit $lines "\n" | safeHTML }}
