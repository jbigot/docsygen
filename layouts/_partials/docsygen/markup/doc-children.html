What a group leaves of a paragraph, and each variant of the copy, renders as that paragraph would, in its place among the parent's children. Inside a variant (.sub.ctx.variant), and without tabpane, the children render in sequence. "group" -}} {{- partial "docsygen/markup/doc-variants.html" (dict "ctx" $.sub.ctx "group" .) -}} {{- $print := merge $.sub (dict "ctx" (merge $.sub.ctx (dict "variant" true "noids" true)) "mdheadings" false) -}}
{{- range .variants -}} {{- $i := .index -}} {{- range .nodes -}} {{- partial "docsygen/markup/doc-node.html" (merge $print (dict "node" . "parent" (partial "docsygen/markup/doc-children/in.html" (dict "parent" $parent "index" $i "node" .)) "index" $i)) -}} {{- end -}} {{- end -}}
{{- else -}} {{- $node := .node -}} {{- if not (and (reflect.IsMap $node) (in $skip $node.tag)) -}} {{- partial "docsygen/markup/doc-node.html" (merge $.sub (dict "node" $node "parent" (partial "docsygen/markup/doc-children/in.html" (dict "parent" $parent "index" .first "node" $node)) "index" .first)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
