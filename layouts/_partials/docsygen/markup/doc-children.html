The example variants among them (docsygen/markup/variants.html) render their children in sequence, as the others. Usage: partial "docsygen/markup/doc-children.html" (dict "parent" $node "sub" (dict "ctx" $ctx …) "skip" (slice "title")) */ -}} {{- $parent := .parent -}} {{- $skip := .skip | default slice -}} {{- $done := -1 -}} {{- range partial "docsygen/markup/variants.html" (dict "ctx" .sub.ctx "children" $parent.children) -}} {{- range $i := seq .first .last -}} {{- if gt $i $done -}} {{- $done = $i -}} {{- $c := index $parent.children $i -}} {{- if not (and (reflect.IsMap $c) (in $skip $c.tag)) -}} {{- partial "docsygen/markup/doc-node.html" (merge $.sub (dict "node" $c "parent" $parent "index" $i)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
