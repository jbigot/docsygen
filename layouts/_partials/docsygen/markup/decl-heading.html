{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* A member's declaration, for the heading of its documentation: inline HTML, on one line, safe in a Markdown heading (no character that Markdown reads as markup), in a . Not necessarily valid C++: the name first, in , then its own template parameters, its parameters and qualifiers, and its type last ("-> T" for a function, ": T" for a variable or an enum's base); a typedef as "Name = definition"; a macro as its name and parameters. No kind keyword, nor what the member's labels say (virtual, static, noexcept, override, …: docsygen/markup/block-member.html), nor the enclosing scope, nor an initializer (the documentation shows it, docsygen/markup/member-doc.html). What follows the name is highlighted inline as C++ (transform.Highlight), with Chroma's classes or styles as the site's markup.highlight asks; the carries Chroma's class, its background and colour left to the heading's. Usage: partial "docsygen/markup/decl-heading.html" (dict "ctx" $ctx "id" $memberId) */ -}} {{- $e := index .ctx.model.ids .id -}} {{- $kind := $e.kind -}} {{- $parts := dict -}} {{- range $e.node.children -}}{{- if and (reflect.IsMap .) (not (isset $parts .tag)) -}}{{- $parts = merge $parts (dict .tag .) -}}{{- end -}}{{- end -}} {{- /* Template arguments spaced as C++ writes them: "Box< T >" reads "Box". */ -}} {{- $tight := `\s*([<>])\s*` -}} {{- $text := dict -}} {{- range slice "name" "type" "argsstring" -}} {{- $t := "" -}} {{- with index $parts . -}}{{- $t = strings.TrimSpace (replaceRE `\s+` " " (partial "docsygen/text/plain.html" .)) -}}{{- end -}} {{- $text = merge $text (dict . $t) -}} {{- end -}} {{- $name := $text.name -}} {{- if hasPrefix $name "operator" -}} {{- /* An operator's name keeps its symbol; a conversion operator's type is spaced as types are. */ -}} {{- if findRE `^operator\s+\w` $name 1 -}}{{- $name = replaceRE `\s+>` ">" (replaceRE `<\s+` "<" $name) -}}{{- end -}} {{- else -}} {{- $name = replaceRE $tight "$1" $name -}} {{- end -}} {{- /* The type without the keywords that the labels say. */ -}} {{- $type := $text.type -}} {{- range seq 4 -}} {{- $type = replaceRE `^(?:virtual|static|inline|explicit|constexpr|consteval|constinit|extern|mutable|friend)\s+` "" $type -}} {{- end -}} {{- $type = replaceRE `\s+>` ">" (replaceRE `<\s+` "<" $type) -}} {{- /* The member's own template parameters, but in a specialization, whose name holds its arguments. */ -}} {{- $tparams := "" -}} {{- if not (strings.Contains $text.name "<") -}} {{- $texts := slice -}} {{- range partial "docsygen/plan/list/params.html" $e.node -}}{{- $texts = $texts | append .text -}}{{- end -}} {{- with $texts -}}{{- $tparams = printf "<%s>" (delimit . ", ") -}}{{- end -}} {{- end -}} {{- $rest := "" -}} {{- if or (in (slice "function" "slot" "signal" "prototype") $kind) (and (eq $kind "friend") (hasPrefix $text.argsstring "(")) -}} {{- $args := partial "docsygen/plan/argsstring.html" $text.argsstring -}} {{- /* What follows the parameters, without what the labels say. */ -}} {{- $qualifiers := strings.TrimSpace (replaceRE `\s*(=\s*0|=\s*delete|=\s*default|\boverride\b|\bfinal\b|\bnoexcept(\([^)]*\))?)\s*` " " $args.rest) -}} {{- $rest = printf "%s%s" $tparams $args.params -}} {{- with $qualifiers -}}{{- $rest = printf "%s %s" $rest . -}}{{- end -}} {{- /* A trailing return type is the type already. */ -}} {{- if and $type (not (and (eq $type "auto") (strings.Contains $qualifiers "->"))) -}}{{- $rest = printf "%s -> %s" $rest $type -}}{{- end -}} {{- else if eq $kind "typedef" -}} {{- /* A function pointer's type ends in its argsstring: int(*)(void *data). */ -}} {{- $rest = printf "%s = %s%s" $tparams (strings.TrimPrefix "typedef " (strings.TrimPrefix "using " $type)) $text.argsstring -}} {{- else if eq $kind "define" -}} {{- $params := slice -}} {{- range $e.node.children -}}{{- if and (reflect.IsMap .) (eq .tag "param") -}}{{- $params = $params | append (partial "docsygen/text/plain.html" .) -}}{{- end -}}{{- end -}} {{- with $params -}}{{- $rest = printf "(%s)" (delimit . ", ") -}}{{- end -}} {{- else if eq $kind "friend" -}} {{- /* A friend class: its type is its kind's keyword. */ -}} {{- $rest = $tparams -}} {{- else -}} {{- /* A variable, an enum, and any other kind: an array's or a function pointer's type ends in its argsstring. */ -}} {{- $rest = $tparams -}} {{- $full := printf "%s%s" $type $text.argsstring -}} {{- with $full -}}{{- $rest = printf "%s: %s" $rest . -}}{{- end -}} {{- end -}} {{- $html := "" -}} {{- with $rest -}} {{- /* transform.Highlight's inline form, its wrapper left out. */ -}} {{- $html = replaceRE `^]*>|$` "" (strings.TrimSpace (replace (string (transform.Highlight . "cpp" "hl_inline=true")) "\n" "")) -}} {{- end -}} {{- $html = printf `%s%s` (htmlEscape $name) $html -}} {{- /* Markdown's punctuation as character references: in a heading, goldmark would read it. The markup's own attributes hold none of these. */ -}} {{- range $c, $r := dict "\\" "\" "*" "*" "_" "_" "`" "`" "[" "[" "]" "]" "{" "{" "}" "}" "~" "~" "$" "$" -}} {{- $html = replace $html $c $r -}} {{- end -}} {{- return safeHTML $html -}}
