{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* One part of a version's reference: the block of address .address (docsygen/part/address.html) through its markup, docsygen/markup/block-.html, with .ctx, a context of that version (docsygen/ctx/root.html). The block is looked for in the plan of .ctx.key, the page being rendered, then in that of the page that holds it: the page of its id (paths.targets), or for a list the page with its role. Its heading is left to the caller. With .anchor, a member block carries its member's anchor on its wrapper. A block embedded away from its own page is away: a description is then whole, its sections in it rather than blocks of their own. A part that the version lacks stops the build. Usage: partial "docsygen/part.html" (dict "address" $address "ctx" $ctx "anchor" false) */ -}} {{- $ctx := .ctx -}} {{- $a := .address -}} {{- $paths := $ctx.paths -}} {{- $keys := slice -}} {{- with $ctx.key -}}{{- $keys = $keys | append . -}}{{- end -}} {{- with $a.id -}} {{- with index $paths.targets . -}}{{- $keys = $keys | append .page -}}{{- end -}} {{- else -}} {{- range $paths.pages -}} {{- $key := .key -}} {{- range .roles -}} {{- if and (eq .role $a.type) (eq (.name | default "") ($a.name | default "")) -}} {{- $keys = $keys | append $key -}} {{- end -}} {{- end -}} {{- end -}} {{- with $a.key -}}{{- $keys = $keys | append . -}}{{- end -}} {{- end -}} {{- $found := false -}} {{- $holder := "" -}} {{- range $keys -}} {{- $key := . -}} {{- if not $found -}} {{- range where $paths.pages "key" $key -}} {{- range partial "docsygen/plan.html" (dict "entry" . "ctx" $ctx) -}} {{- if and (not $found) (eq (partial "docsygen/part/address.html" .) $a) -}}{{- $found = . -}}{{- $holder = $key -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* A compound's brief is the lead of its page, not a block of its plan. */ -}} {{- if and (not $found) (eq $a.type "brief") $a.id -}} {{- with index $paths.targets $a.id -}} {{- $found = dict "type" "brief" "id" $a.id "part" "brief" -}}{{- $holder = .page -}} {{- end -}} {{- end -}} {{- /* A class's list of all members is in no page's plan: it is for hand-written pages. */ -}} {{- if and (not $found) (eq $a.type "all-members") $a.id -}} {{- with index $paths.targets $a.id -}} {{- $found = dict "type" "all-members" "id" $a.id -}}{{- $holder = .page -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "docsygen: no part %s in version %s" (jsonify $a) $ctx.version -}} {{- return "" -}} {{- end -}} {{- $onPage := false -}} {{- with $ctx.page -}}{{- $onPage = true -}}{{- end -}} {{- $away := and $onPage (ne ($ctx.key | default "") $holder) -}} {{- $b := merge $found (dict "heading" false "anchor" (and .anchor (eq $found.type "member") true) "away" $away "whole" (and $away (eq $found.type "description"))) -}} {{- $markup := printf "docsygen/markup/block-%s.html" $b.type -}} {{- if not (templates.Exists (printf "_partials/%s" $markup)) -}} {{- warnidf (printf "docsygen-block-%s" $b.type) "docsygen: no markup for the block type %q" $b.type -}} {{- return "" -}} {{- end -}} {{- return partial $markup (dict "block" $b "ctx" (merge $ctx (dict "key" $holder))) -}}
