{{- /* The class hierarchy (writeClassHierarchy): the root classes sorted as Doxygen sorts its class list (qstricmp_sort on the scopeless name, then on the full name), each with its derived classes, a class expanded only once. Doxygen names an undocumented base without its template arguments, unless a reference carries its refid. Returns [{id, link, name, kind, children}], id "" for the nodes without a description. Usage: partial "docsygen/plan/tree/hierarchy.html" (dict "model" $model) */ -}} {{- $m := .model -}} {{- $classKinds := slice "class" "struct" "union" "interface" "protocol" "category" "exception" -}} $info = dict "id" $c.id "link" $c.id "name" $dn "full" $c.name -}} {{- else -}} {{- $key = printf "inst:%s" .text -}} {{- $info = dict "id" "" "link" $c.id "name" .text "full" .text -}} {{- end -}} {{- else -}} {{- $name := .text -}} {{- if not (index (.s.Get "withRefid") .text) -}}{{- $name = trim (replaceRE `<.*$` "" .text) " " -}}{{- end -}} {{- $key = printf "out:%s" $name -}} {{- $info = {{- end -}} {{- return $key -}} {{- end -}} {{- define "_partials/docsygen/plan/tree/hierarchy-add.html" -}} {{- /* Appends .value to .s's list .list[.key], once. */ -}} {{- $l := index (.s.Get .list)
