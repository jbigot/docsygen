(eq .tag $tag) -}}{{- $listed.Set .attributes.refid true -}}{{- end -}}{{- end -}} {{- end -}} {{- range $m.order -}} {{- $e := index $m.ids . -}} {{- if and (eq $e.kind $kind) (ne . "indexpage") (not ($listed.Get .)) -}} {{- $nodes = $nodes | append (partial "docsygen/plan/tree/titled.html" (dict "model" $m "id" . "tag" $tag "kind" $kind)) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .name "files" -}} {{- /* The directories no other one lists, each with its directories then its files, then the files outside any directory (writeDirHierarchy). */ -}} {{- $listed := newScratch -}} {{- range $m.order -}} {{- range (index $m.ids .).node.children -}}{{- if and (reflect.IsMap .) "name" $e.name "kind" "example" "children" slice) -}}{{- end -}} {{- end -}} {{- end -}} {{- return $nodes -}}
