Usage: partial "docsygen/plan/title.html" (dict "model" $model "id" $refid) */ -}} {{- $m := .model -}} {{- $e := index $m.ids .id -}} {{- $hide := eq (index ((index $m.options "HIDE_COMPOUND_REFERENCE") | default (slice "NO")) 0) "YES" -}} {{- $classLike := slice "class" "struct" "union" "interface" "protocol" "category" "exception" -}} {{- $name := partial "docsygen/plan/display-name.html" (dict "model" $m "id" .id) -}} {{- $text := $name -}} {{- $labels := slice -}} {{- $title := "" -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "title") -}}{{- $title = partial "docsygen/text/plain.html" . -}}{{- end -}} {{- end -}} {{- if in $classLike $e.kind -}} {{- $template := false -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "templateparamlist") -}} {{- range .children -}}{{- if reflect.IsMap . -}}{{- $template = true -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $hide -}}{{- $text = T (printf "docsygen-%s%s-reference" $e.kind (cond $template "-template" "")) (dict "name" $name) -}}{{- end -}} {{- $a := $e.node.attributes -}} {{- range slice "final" "sealed" "abstract" "export" -}} {{- if eq (index $a .) "yes" -}}{{- $labels = $labels | append . -}}{{- end -}} {{- end -}} {{- range $e.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "location") -}}{{- $name = .attributes.file -}}{{- end -}} {{- end -}} {{- $text = cond $hide $name $text -}} {{- end -}} {{- if $suffix -}} {{- if not $hide -}}{{- $text = T (printf "docsygen-%s-reference" $suffix) (dict "name" $name) -}}{{- end -}} {{- else if $title -}} {{- $text = $title -}} {{- end -}} {{- end -}} {{- return dict "text" $text "labels" $labels -}}
