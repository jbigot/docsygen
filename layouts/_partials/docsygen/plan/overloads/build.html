{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The overload sets (docsygen/plan/overloads.html) of every compound of a model, as {refid: sets}, built once per model, in one partialCached entry like docsygen/plan/sections/all.html. Only the compounds with a set are in it. */ -}} {{- $m := . -}} {{- /* Scratch maps: merge would fold the case of refids. */ -}} {{- $all := newScratch -}} {{- $all.Set "m" dict -}} {{- range $cid := $m.order -}} {{- $out := newScratch -}} {{- range partial "docsygen/plan/sections.html" (dict "model" $m "id" $cid) -}} {{- if eq .group "summary" -}} {{- $g := partial "docsygen/plan/overloads/group.html" (dict "model" $m "members" .members) -}} {{- $keys := $g.keys -}} {{- $others := $g.others -}} {{- $marked := and (hasPrefix .key "user-defined-") (not $others) (eq (len $keys) 1) -}} {{- $description := cond $marked .description false -}} {{- $group := cond $marked .key "" -}} {{- range $keys -}} {{- $members := index $g.by . -}} {{- if ge (len $members) 2 -}} {{- $set := dict "set" (index $members 0) "members" $members "description" ($description | default false) "group" $group -}} {{- range $members -}}{{- $out.Set . $set -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $out.Values -}}{{- $all.SetInMap "m" $cid . -}}{{- end -}} {{- end -}} {{- return $all.Get "m" -}}
