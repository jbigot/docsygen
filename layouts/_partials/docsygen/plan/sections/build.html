{{- /* SPDX-FileCopyrightText: 2026 Julien Bigot SPDX-License-Identifier: MIT */ -}} {{- /* The body of docsygen/plan/sections.html, with the sections already built of other compounds in .known ({refid: sections}); the others a file needs are built here. */ -}} {{- $m := .model -}} {{- $e := index $m.ids .id -}} {{- $kind := $e.kind -}} {{- $private := eq (index ((index $m.options "EXTRACT_PRIVATE") | default (slice "NO")) 0) "YES" -}} {{- $classLike := slice "class" "struct" "union" "interface" "protocol" "category" "exception" -}} {{- $isClass := in $classLike $kind -}} {{- /* Gather the members of each sectiondef, the inner compounds and the member groups. */ -}} {{- $found := newScratch -}} {{- $groups := slice -}} {{- range $e.node.children -}} {{- if not (reflect.IsMap .) -}} {{- else if in (slice "innerclass" "innernamespace" "innerconcept" "innergroup" "innerdir" "innerfile") .tag -}} {{- $target := index $m.ids .attributes.refid -}} {{- if and $target (or $target.name (ne $target.kind "namespace")) (or $private (ne .attributes.prot "private")) -}} {{- $key := printf "inner-%s" (strings.TrimPrefix "inner" .tag) -}} {{- $found.Set $key (($found.Get $key | default slice) | append .attributes.refid) -}} {{- end -}} {{- else if eq .tag "sectiondef" -}} {{- $ids := slice -}} {{- $header := "" -}} {{- $description := false -}} {{- range .children -}} {{- if not (reflect.IsMap .) -}} {{- else if in (slice "memberdef" "member") .tag -}} {{- $mid := .attributes.id | default .attributes.refid -}} {{- if partial "docsygen/model/linkable.html" (dict "model" $m "id" $mid) -}}{{- $ids = $ids | append $mid -}}{{- end -}} {{- else if eq .tag "header" -}} {{- $header = partial "docsygen/text/plain.html" . -}} {{- else if eq .tag "description" -}} {{- $description = . -}} {{- end -}} {{- end -}} {{- if $ids -}} {{- if eq .attributes.kind "user-defined" -}} {{- $groups = $groups | append (dict "heading" $header "description" $description "members" $ids) -}} {{- else -}} {{- $found.Set .attributes.kind (($found.Get .attributes.kind | default slice) | append $ids) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* A file page also lists the members it declares that another compound documents, in their member group when they have one there. */ -}} {{- if eq $kind "file" -}} {{- $byKind := dict "function" "func" "variable" "var" "typedef" "typedef" "enum" "enum" "define" "define" -}} {{- $inGroup := newScratch -}} {{- range index $m.fileMembers .id -}} {{- $mid := . -}} {{- $home := (index $m.ids $mid).home -}} {{- $homeSections := index $.known $home -}} {{- if not $homeSections -}}{{- $homeSections = partial "docsygen/plan/sections/build.html" (dict "model" $m "id" $home "known" $.known) -}}{{- end -}} {{- range $homeSections -}} {{- if and (hasPrefix .key "user-defined-") (in .members $mid) -}}{{- $inGroup.Set $mid . -}}{{- end -}} {{- end -}} {{- with $inGroup.Get $mid -}} {{- $group := . -}} {{- $found.Set "file-groups" (($found.Get "file-groups" | default slice) | append (dict "heading" $group.heading "description" $group.description "member" $mid)) -}} {{- else -}} {{- with index $byKind (index $m.ids $mid).kind -}} {{- if and (partial "docsygen/model/linkable.html" (dict "model" $m "id" $mid)) (not (in ($found.Get . | default slice) $mid)) -}}{{- $found.Set . (($found.Get . | default slice) | append $mid) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $found.Get "file-groups" -}} {{- $g := . -}} {{- /* Shown in its group only. */ -}} {{- range $k, $list := $found.Values -}} {{- if and (reflect.IsSlice $list) (in $list $g.member) (ne $k "file-groups") -}}{{- $found.Set $k (complement (slice $g.member) $list) -}}{{- end -}} {{- end -}} {{- $known := false -}} {{- $merged := slice -}} {{- range $groups -}} {{- if eq .heading $g.heading -}}{{- $known = true -}}{{- $merged = $merged | append (merge . (dict "members" (cond (in .members $g.member) .members (.members | append $g.member)))) -}} {{- else -}}{{- $merged = $merged | append . -}}{{- end -}} {{- end -}} {{- if not $known -}}{{- $merged = $merged | append (dict "heading" $g.heading "description" $g.description "members" (slice $g.member)) -}}{{- end -}} {{- $groups = $merged -}} {{- end -}} {{- end -}} {{- /* The summary sections, in the layout's order. */ -}} {{- $order := slice -}} {{- if $isClass -}} {{- $order = slice "inner-class" "public-type" "public-slot" "signal" "public-func" "public-static-func" "public-attrib" "public-static-attrib" "protected-type" "protected-slot" "protected-func" "protected-static-func" "protected-attrib" "protected-static-attrib" "package-type" "package-func" "package-static-func" "package-attrib" "package-static-attrib" "property" "event" "private-type" "private-slot" "private-func" "private-static-func" "private-attrib" "private-static-attrib" "friend" "related" -}} {{- else if eq $kind "namespace" -}} {{- $order = slice "inner-namespace" "inner-class" "inner-concept" "typedef" "enum" "func" "var" "property" -}} {{- else if eq $kind "file" -}} {{- $order = slice "inner-class" "inner-namespace" "inner-concept" "define" "typedef" "enum" "func" "var" "property" -}} {{- else if eq $kind "group" -}} {{- $order = slice "inner-group" "inner-dir" "inner-file" "inner-namespace" "inner-concept" "inner-class" "define" "typedef" "enum" "enumvalue" "func" "var" "signal" "public-slot" "protected-slot" "private-slot" "event" "property" "friend" -}} {{- else if eq $kind "dir" -}} {{- $order = slice "inner-dir" "inner-file" -}} {{- end -}} {{- $sections := slice -}} {{- range $order -}} {{- $key := . -}} {{- $list := $found.Get $key -}} {{- if $list -}} {{- $inner := hasPrefix $key "inner-" -}} {{- $sections = $sections | append (dict "key" $key "group" "summary" "members" (cond $inner slice $list) "inner" (cond $inner $list slice)) -}} {{- end -}} {{- end -}} {{- range $i, $g := $groups -}} {{- $key := printf "user-defined-%d" (add $i 1) -}} {{- $sections = $sections | append (dict "key" $key "heading" $g.heading "description" $g.description "group" "summary" "members" $g.members "inner" slice) -}} {{- end -}} {{- /* The detailed sections, in the layout's order: every member that has a detailed entry, from the summary sections and the member groups, but not those another compound documents. */ -}} {{- $doc := newScratch -}} {{- $short := replaceRE `<.*$` "" (index (split $e.name "::") (sub (len (split $e.name "::")) 1)) -}} {{- range $sections -}} {{- $section := .key -}} {{- range .members -}} {{- $member := index $m.ids . -}} {{- if and (eq $member.home $.id) (partial "docsygen/plan/detailed.html" (dict "model" $m "id" .)) -}} {{- $name := index (split $member.name "::") (sub (len (split $member.name "::")) 1) -}} {{- $key := "" -}} {{- if eq $member.kind "friend" -}}{{- $key = "doc-friend" -}} {{- else if eq $section "related" -}}{{- $key = "doc-related" -}} {{- else if in (slice "function" "slot" "signal") $member.kind -}}{{- $key = "doc-function" -}} {{- else if eq $member.kind "variable" -}}{{- $key = "doc-variable" -}} {{- else if in (slice "typedef" "enum" "define" "property" "event") $member.kind -}}{{- $key = printf "doc-%s" $member.kind -}} {{- end -}} {{- if $key -}}{{- $doc.Set $key (($doc.Get $key | default slice) | append .) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $docOrder := cond $isClass (slice "doc-typedef" "doc-enum" "doc-function" "doc-friend" "doc-related" "doc-variable" "doc-property" "doc-event") (slice "doc-define" "doc-typedef" "doc-enum" "doc-function" "doc-variable" "doc-property" "doc-event") -}} {{- /* Each detailed section in Doxygen's order (docsygen/plan/sections/sort.html). */ -}} {{- range $docOrder -}} {{- $key := . -}} {{- with $doc.Get $key -}} {{- $doc.Set $key (partial "docsygen/plan/sections/sort.html" (dict "model" $m "ids" . "class" $isClass)) -}} {{- end -}} {{- with $doc.Get $key -}} {{- $sections = $sections | append (dict "key" $key "heading" (partial "docsygen/plan/section-heading.html" (dict "model" $m "key" $key "kind" $kind)) "group" "doc" "members" . "inner" slice) -}} {{- end -}} {{- end -}} {{- return $sections -}}
