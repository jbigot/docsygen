Usage: partial "docsygen/plan/sections/sort.html" (dict "model" $model "ids" $ids "class" false) */ -}} {{- $m := .model -}} {{- $sort := eq (index ((index $m.options "SORT_MEMBER_DOCS") | default (slice "YES")) 0) "YES" -}} {{- $out := .ids -}} {{- if or $sort .class -}} {{- $keyed := slice -}} {{- range $id := .ids -}} {{- $member := index $m.ids $id -}} {{- $name := index (split $member.name "::") (sub (len (split $member.name "::")) 1) -}} {{- $line := 0 -}} {{- $column := 0 -}} {{- range $member.node.children -}} {{- if and (reflect.IsMap .) (eq .tag "location") -}}{{- $line = int (.attributes.line | default 0) -}}{{- $column = int (.attributes.column | default 0) -}}{{- end -}} {{- end -}} {{- $keyed = $keyed | append (dict "id" $id "key" (cond $sort $name (add (mul $line 1000000) $column))) -}} {{- end -}} {{- $out = slice -}} {{- range sort $keyed "key" -}}{{- $out = $out | append .id -}}{{- end -}} {{- end -}} {{- return $out -}}
