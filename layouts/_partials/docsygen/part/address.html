{{- /* A block's address, what names it among the blocks of a version (docsygen/part.html): its type, id, key, name, filter and letter, those that are set, as strings. The heading and the lists of a block (ids) are not part of it: a block's other fields derive from these. Usage: partial "docsygen/part/address.html" $block */ -}} {{- $a := dict -}} {{- range $f := slice "type" "id" "key" "name" "filter" "letter" -}} {{- with index $ $f -}}{{- $a = merge $a (dict $f (string .)) -}}{{- end -}} {{- end -}} {{- return $a -}}
