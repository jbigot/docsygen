{{- /* The body of docsygen/versions.html. The roots of the versions that development mode skips, and the pages that mark the calls marked latest (docsygen/adapter.html), are in no page collection but their parent section's, which holds the other roots too. Two calls marked latest both generate latest/, where one root wins: their marks tell them apart. */ -}} {{- $roots := where site.Pages "Params.docsygen.root" true -}} {{- $marks := slice -}} {{- range $roots -}} {{- with .Parent -}} {{- $roots = $roots | append (where (where .Pages "Params.docsygen.root" true) "Params.docsygen.skipped" true) -}} {{- $marks = $marks | append (where .Pages "Params.docsygen.latestMark" true) -}} {{- end -}} {{- end -}} {{- $latestLabels := slice -}} {{- range $marks -}}{{- $latestLabels = $latestLabels | append .Params.docsygen.version -}}{{- end -}} {{- $latestLabels = uniq $latestLabels -}} {{- if gt (len $latestLabels) 1 -}} {{- errorf "docsygen: versions %s are all marked latest: mark one" (delimit (sort $latestLabels) ", ") -}} {{- end -}} {{- $byLabel := newScratch -}} {{- $labels := slice -}} {{- range sort $roots "Params.docsygen.call" -}} {{- $p := .Params.docsygen -}} {{- if not ($byLabel.Get $p.version) -}} {{- $labels = $labels | append $p.version -}} {{- $byLabel.Set $p.version (dict "version" $p.version "dir" $p.dir "latest" ($p.latest | default false) "release" ($p.release | default false) "path" $p.path "page" . "pathMap" ($p.pathMap | default slice) "search" ($p.search | default false) "skipped" ($p.skipped | default false)) -}} {{- end -}} {{- end -}} {{- with site.Params.docsygen -}}{{- with .versions -}} {{- range . -}} {{- if not (or (eq (string .) "latest") (in $labels (string .))) -}} {{- errorf "docsygen: params.docsygen.versions lists %q, which no adapter call has (YAML reads an unquoted 1.10 as 1.1: quote version numbers)" (string .) -}} {{- end -}} {{- end -}} {{- end -}}{{- end -}} {{- $out := slice -}} {{- range partial "docsygen/versions/order.html" $labels -}}{{- $out = $out | append ($byLabel.Get .) -}}{{- end -}} {{- return $out -}}
