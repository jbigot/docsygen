{{- /* What the renderers of a generated page need (docsygen/ctx/root.html). The version root is the page itself or its nearest ancestor with params.docsygen.root (never derived from .Path, which Hugo lowercases). Usage: partial "docsygen/ctx.html" .Page */ -}} {{- $root := false -}} {{- if .Params.docsygen.root -}} {{- $root = . -}} {{- else -}} {{- range .Ancestors -}}{{- if and (not $root) .Params.docsygen.root -}}{{- $root = . -}}{{- end -}}{{- end -}} {{- end -}} {{- $ctx := dict -}} {{- if not $root -}} {{- errorf "docsygen: %s is not under a docsygen version root" .Path -}} {{- else -}} {{- $ctx = partial "docsygen/ctx/root.html" (dict "root" $root "page" .) -}} {{- end -}} {{- return $ctx -}}
