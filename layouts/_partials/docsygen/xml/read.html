{{- /* The body of docsygen/xml/unmarshal.html: the template reader on the resource's content. The tree is kept in hugo.Store, under the resource's name and with a hash of its content, so a resource is parsed once, and a hugo server rebuild re-parses only the files that changed. Not partialCached: its LRU of 1000 entries would evict the models for these. */ -}} {{- $key := printf "docsygen/xml|%s" .Name -}} {{- $content := .Content -}} {{- $hash := md5 $content -}} {{- $entry := hugo.Store.Get $key -}} {{- if or (not $entry) (ne $entry.hash $hash) -}} {{- $entry = dict "hash" $hash "tree" (partial "docsygen/xml/parse.html" $content) -}} {{- hugo.Store.Set $key $entry -}} {{- end -}} {{- return $entry.tree -}}
