{{- /* Block n of the page's plan (docsygen/plan.html), through its flavor's markup, docsygen/markup/block-.html, with the same block dict in both flavors. Its heading is set to false: the page's Markdown skeleton (docsygen/skeleton.html) wrote it. Usage, in the skeleton: {{< docsygen/block n=3 >}} */ -}} {{- $ctx := partial "docsygen/ctx.html" .Page -}} {{- $key := .Page.Params.docsygen.key -}} {{- $n := int (.Get "n") -}} {{- range where $ctx.paths.pages "key" $key -}} {{- $blocks := partial "docsygen/plan.html" (dict "entry" . "ctx" $ctx) -}} {{- if ge $n (len $blocks) -}} {{- errorf "docsygen: %s has no block %d" $.Page.Path $n -}} {{- else -}} {{- $b := merge (index $blocks $n) (dict "heading" false) -}} {{- $markup := printf "docsygen/markup/block-%s.html" $b.type -}} {{- if templates.Exists (printf "_partials/%s" $markup) -}} {{- partial $markup (dict "block" $b "ctx" $ctx) -}} {{- else -}} {{- warnidf (printf "docsygen-block-%s" $b.type) "docsygen: no markup for the block type %q" $b.type -}} {{- end -}} {{- end -}} {{- end -}}
